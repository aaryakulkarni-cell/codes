#include <iostream>
#include <vector>
#include <cstdlib>
#include <ctime>
using namespace std;

// Partition using two pointers
int partition(vector<int>& a, int low, int high, int& comp, int& swaps)
{
    int pivot = a[low];

    int i = low + 1;
    int j = high;

    while (i <= j)
    {
        while (i <= high && a[i] <= pivot)
        {
            comp++;
            i++;
        }

        while (j > low && a[j] > pivot)
        {
            comp++;
            j--;
        }

        if (i < j)
        {
            swap(a[i], a[j]);
            swaps++;
        }
    }

    swap(a[low], a[j]);
    swaps++;

    return j;
}

// Deterministic Quick Sort
void deterministicQuickSort(vector<int>& a, int low, int high,
                            int& comp, int& swaps)
{
    if (low < high)
    {
        int p = partition(a, low, high, comp, swaps);

        deterministicQuickSort(a, low, p - 1, comp, swaps);
        deterministicQuickSort(a, p + 1, high, comp, swaps);
    }
}

// Randomized Quick Sort
void randomizedQuickSort(vector<int>& a, int low, int high,
                         int& comp, int& swaps)
{
    if (low < high)
    {
        int randomIndex = low + rand() % (high - low + 1);

        swap(a[low], a[randomIndex]);
        swaps++;

        int p = partition(a, low, high, comp, swaps);

        randomizedQuickSort(a, low, p - 1, comp, swaps);
        randomizedQuickSort(a, p + 1, high, comp, swaps);
    }
}

// Display array
void display(vector<int>& a)
{
    for (int x : a)
        cout << x << " ";

    cout << endl;
}

int main()
{
    int n;

    cout << "Enter number of elements: ";
    cin >> n;

    if (n <= 0)
    {
        cout << "Invalid array size";
        return 0;
    }

    vector<int> a(n);

    cout << "Enter array elements: ";
    for (int i = 0; i < n; i++)
        cin >> a[i];

    vector<int> b = a;

    int comp1 = 0, swaps1 = 0;
    int comp2 = 0, swaps2 = 0;

    srand(time(0));

    deterministicQuickSort(a, 0, n - 1, comp1, swaps1);
    randomizedQuickSort(b, 0, n - 1, comp2, swaps2);

    cout << "\nDeterministic Quick Sort:\n";
    display(a);
    cout << "Comparisons: " << comp1 << endl;
    cout << "Swaps: " << swaps1 << endl;

    cout << "\nRandomized Quick Sort:\n";
    display(b);
    cout << "Comparisons: " << comp2 << endl;
    cout << "Swaps: " << swaps2 << endl;

    return 0;
}
