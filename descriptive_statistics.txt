#Part 1 - 
import pandas as pd

data = {
    'gender': ['male','female','male','female','male'],
    'marks': [80, 90, 70, 85, 60]
}

df = pd.DataFrame(data)

df

# Grouped statistics
df.groupby('gender')['marks'].describe()

# List of values per category (IMPORTANT)
df.groupby('gender')['marks'].apply(list)
